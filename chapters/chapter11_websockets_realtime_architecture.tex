\section{HTTP Polling vs WebSockets Architectural Evaluation}
In high-concurrency tournament environments, traditional HTTP polling introduces substantial overhead:

\begin{table}[htbp]
\centering
\footnotesize
\begin{tabular}{|l|p{5.5cm}|p{5.8cm}|}
\hline
\rowcolor{primaryBlue!15}
\textbf{Architectural Metric} & \textbf{HTTP Long-Polling} & \textbf{SignalR / WebSockets} \\ \hline
\textbf{Connection Lifecycle} & New TCP/TLS handshake on every poll ($200-400\text{ms}$). & Single persistent full-duplex TCP socket. \\ \hline
\textbf{Server Network Load} & 100,000 requests/sec with heavy HTTP headers. & Minimal heartbeat frames ($< 32\text{ bytes}$ per frame). \\ \hline
\textbf{End-to-End Latency} & $500\text{ms} - 2000\text{ms}$ polling interval lag. & $\mathbf{< 15\text{ms}}$ instantaneous push delivery. \\ \hline
\textbf{Live Head-to-Head} & Imposible to synchronize microsecond tie-breaks. & Perfectly synchronized server-authoritative timer. \\ \hline
\end{tabular}
\caption{Architectural Comparison: HTTP Polling vs Full-Duplex WebSockets}
\end{table}

\section{.NET 10 SignalR Hub Implementation}

\begin{lstlisting}[language={[Sharp]C}, caption={Live Tournament Duel SignalR Hub in .NET 10}]
[Authorize]
public class LiveDuelHub : Hub
{
    private readonly IDuelMatchmaker _matchmaker;
    private readonly ILogger<LiveDuelHub> _logger;

    public LiveDuelHub(IDuelMatchmaker matchmaker, ILogger<LiveDuelHub> logger)
    {
        _matchmaker = matchmaker;
        _logger = logger;
    }

    public async Task JoinQueue(int eventId)
    {
        var msisdn = Context.UserIdentifier!;
        var match = await _matchmaker.EnqueuePlayerAsync(msisdn, eventId, Context.ConnectionId);

        if (match != null)
        {
            // Both players matched: notify room instantly
            await Clients.Clients(match.PlayerAConnection, match.PlayerBConnection)
                .SendAsync("MatchFound", match);
        }
    }

    public async Task SubmitLiveAnswer(string matchId, int questionId, string selectedOption)
    {
        var result = await _matchmaker.RecordLiveAnswerAsync(matchId, Context.UserIdentifier!, questionId, selectedOption);
        
        // Broadcast opponent progress bar update (< 15ms latency)
        await Clients.Group(matchId).SendAsync("OpponentProgressUpdated", result);
    }
}
\end{lstlisting}
